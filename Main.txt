--// UTILITY
local function safeCall(fn, ...)
    local args = table.pack(...)

    local success, result = pcall(function()
        return fn(table.unpack(args, 1, args.n))
    end)

    if success then
        return result
    end

    return nil
end

local function tween(instance, properties, duration, style, direction)
    if not settings.animations then
        for property, value in pairs(properties) do
            instance[property] = value
        end

        return
    end

    local info = TweenInfo.new(
        duration or 0.18,
        style or Enum.EasingStyle.Quart,
        direction or Enum.EasingDirection.Out
    )

    local animation = TweenService:Create(instance, info, properties)
    animation:Play()

    return animation
end

local function createCorner(parent, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius or 10)
    corner.Parent = parent

    return corner
end

local function createStroke(parent, color, transparency, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color or COLORS.line
    stroke.Transparency = transparency or 0
    stroke.Thickness = thickness or 1
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = parent

    return stroke
end

local function createGradient(parent, rotation)
    local gradient = Instance.new("UIGradient")
    gradient.Rotation = rotation or 90

    gradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(35, 35, 40)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(17, 17, 21)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(7, 7, 9))
    })

    gradient.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.1),
        NumberSequenceKeypoint.new(0.5, 0.25),
        NumberSequenceKeypoint.new(1, 0.08)
    })

    gradient.Parent = parent

    return gradient
end

local function formatNumber(number)
    if number >= 1000000 then
        return string.format("%.1fM", number / 1000000)
    elseif number >= 1000 then
        return string.format("%.1fk", number / 1000)
    end

    return tostring(number)
end

local function formatTime(seconds)
    local minutes = math.floor(seconds / 60)
    local secs = seconds % 60

    return string.format("%02d:%02d", minutes, secs)
end

local function Toast(message, duration)
    if not toastHolder or not toastHolder.Parent then
        return
    end

    local toast = Instance.new("Frame")
    toast.Size = UDim2.new(0, 240, 0, 38)
    toast.BackgroundColor3 = COLORS.glass2
    toast.BackgroundTransparency = 0.04
    toast.BorderSizePixel = 0
    toast.Position = UDim2.new(1, 15, 1, -60)
    toast.Parent = toastHolder

    createCorner(toast, 10)
    createStroke(toast, COLORS.lineBright, 0.15, 1)

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(0, 3, 1, -10)
    bar.Position = UDim2.new(0, 6, 0, 5)
    bar.BackgroundColor3 = COLORS.white
    bar.BorderSizePixel = 0
    bar.Parent = toast

    createCorner(bar, 3)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -25, 1, 0)
    label.Position = UDim2.new(0, 17, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = tostring(message)
    label.TextColor3 = COLORS.white
    label.TextSize = 11
    label.Font = Enum.Font.GothamMedium
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = toast

    tween(
        toast,
        {
            Position = UDim2.new(1, -250, 1, -60)
        },
        0.25
    )

    task.delay(duration or 2, function()
        if toast and toast.Parent then
            tween(
                toast,
                {
                    Position = UDim2.new(1, 15, 1, -60)
                },
                0.22
            )

            task.wait(0.25)

            if toast then
                toast:Destroy()
            end
        end
    end)
end

local function GetTypingDelay()
    local wpm = math.min(settings.wpm, 700)

    local lettersPerSecond = (wpm * 5) / 60
    local baseDelay = 1 / lettersPerSecond

    return baseDelay * 1000
end

local function HumanDelay()
    if not settings.humanMode then
        return
    end

    local baseDelay = GetTypingDelay()

    local variance = baseDelay * (math.random(-35, 35) / 100)

    local actualDelay = math.max(
        8,
        baseDelay + variance
    )

    task.wait(actualDelay / 1000)
end

local function SimulateTypo(word)
    if settings.riskyMistakes then
        if math.random(1, 100) > settings.mistakeChance * 1.5 then
            return nil
        end
    else
        if math.random(1, 100) > settings.mistakeChance then
            return nil
        end
    end

    if #word < 4 then
        return nil
    end

    local position = math.random(1, #word)

    local wrongLetter = string.char(
        math.random(97, 122)
    )

    local correctLetter = string.sub(
        word,
        position,
        position
    )

    return {
        position = position,
        wrongLetter = wrongLetter,
        correctLetter = correctLetter
    }
end

local function FindAnswerBox(frame)
    if not frame then
        return nil
    end

    local candidates = {
        "TextBox",
        "AnswerBox",
        "Input",
        "ChatInput"
    }

    for _, name in ipairs(candidates) do
        local object = frame:FindFirstChild(name, true)

        if object and object:IsA("TextBox") then
            return object
        end
    end

    for _, object in ipairs(frame:GetDescendants()) do
        if object:IsA("TextBox") and object.Visible then
            return object
        end
    end

    return nil
end

--// TYPE HUMAN
local function TypeWordHuman(word)
    local frame = GetInGameFrame()

    if not frame then
        return false
    end

    local answerBox = FindAnswerBox(frame)

    if not answerBox then
        for i = 1, #word do
            local char = string.sub(word, i, i):lower()

            local keyCode = Enum.KeyCode[char:upper()]

            if keyCode then
                VirtualInput:SendKeyEvent(
                    keyCode,
                    true,
                    false
                )

                HumanDelay()

                VirtualInput:SendKeyEvent(
                    keyCode,
                    false,
                    false
                )

                if i < #word then
                    HumanDelay()
                end
            end
        end

        task.wait(0.05)

        VirtualInput:SendKeyEvent(
            Enum.KeyCode.Return,
            true,
            false
        )

        task.wait(0.02)

        VirtualInput:SendKeyEvent(
            Enum.KeyCode.Return,
            false,
            false
        )

        return true
    end

    answerBox:CaptureFocus()

    task.wait(0.05)

    answerBox.Text = ""

    task.wait(0.05)

    for i = 1, #word do
        local char = string.sub(word, i, i)

        local typo = SimulateTypo(word)

        if typo and typo.position == i then
            answerBox.Text =
                answerBox.Text ..
                typo.wrongLetter

            HumanDelay()

            task.wait(
                GetTypingDelay() / 1000 * 0.4
            )

            VirtualInput:SendKeyEvent(
                Enum.KeyCode.Backspace,
                true,
                false
            )

            task.wait(0.04)

            VirtualInput:SendKeyEvent(
                Enum.KeyCode.Backspace,
                false,
                false
            )

            HumanDelay()

            answerBox.Text =
                answerBox.Text ..
                typo.correctLetter

            mistakesMade += 1
        else
            answerBox.Text =
                answerBox.Text ..
                char
        end

        HumanDelay()
    end

    task.wait(
        GetTypingDelay() / 1000 * 0.6
    )

    VirtualInput:SendKeyEvent(
        Enum.KeyCode.Return,
        true,
        false
    )

    task.wait(0.03)

    VirtualInput:SendKeyEvent(
        Enum.KeyCode.Return,
        false,
        false
    )

    task.wait(0.03)

    answerBox:ReleaseFocus()

    return true
end

    VirtualInput:SendKeyEvent(
        Enum.KeyCode.Return,

local function TypeWord(word)
    if not word or word == "" then
        return
    end

    word = tostring(word):lower()

    local success

    if settings.humanMode then
        success = TypeWordHuman(word)
    else
        success = TypeWordFast(word)
    end

    if success then
        usedWords[word] = true

        wordsTyped += 1
        correctWords += 1

        UpdateFilteredDictionary()
        UpdateSuggestions()
    end
end

--// KILLER WORD
local function IsKillerWord(word)
    local lowerWord = word:lower()

    for _, ending in ipairs(hardEndings) do
        if #ending <= #lowerWord and
            string.sub(
                lowerWord,
                -#ending
            ) == ending then

            return true
        end
    end

    return false
end

--// FILTER
local function FilterWordsByMode()
    local filtered = {}

    local minLen =
        settings.minWordLength or 3

    local maxLen =
        settings.maxWordLength or 20

    for _, word in ipairs(dictionary) do
        local wordLen = #word

        if wordLen >= minLen and
            wordLen <= maxLen then

            if not usedWords[word] or
                settings.showUsedWords then

                if settings.wordFilter == "all" then
                    table.insert(filtered, word)

                elseif settings.wordFilter == "short" and
                    wordLen >= 3 and
                    wordLen <= 5 then

                    table.insert(filtered, word)

                elseif settings.wordFilter == "middle" and
                    wordLen >= 6 and
                    wordLen <= 9 then

                    table.insert(filtered, word)

                elseif settings.wordFilter == "long" and
                    wordLen >= 10 and
                    wordLen <= 20 then

                    table.insert(filtered, word)

                elseif settings.wordFilter == "killer" and
                    IsKillerWord(word) and
                    wordLen >= 3 then

                    table.insert(filtered, word)
                end
            end
        end
    end

    if settings.priorityMode == "speed" then
        table.sort(
            filtered,
            function(a, b)
                return #a < #b
            end
        )

    elseif settings.priorityMode == "killer" then
        table.sort(
            filtered,
            function(a, b)
                local scoreA =
                    IsKillerWord(a) and 1 or 0

                local scoreB =
                    IsKillerWord(b) and 1 or 0

                return scoreA > scoreB
            end
        )
    end

    return filtered
end

UpdateFilteredDictionary = function()
    filteredDictionary =
        FilterWordsByMode()

    prefixCache = {}

    if isInitialized then
        UpdateSuggestions()
    end
end

--// CURRENT WORD
local function GetCurrentWord()
    local frame = GetInGameFrame()

    if not frame then
        return ""
    end

    local currentWordFrame =
        frame:FindFirstChild(
            "CurrentWord",
            true
        )

    if currentWordFrame and
        currentWordFrame.Visible then

        local letters = {}
        local numbered = {}

        for _, child in ipairs(
            currentWordFrame:GetChildren()
        ) do
            local idx = tonumber(child.Name)

            if idx and
                child:FindFirstChild("Letter") then

                table.insert(
                    numbered,
                    {
                        idx = idx,
                        obj = child
                    }
                )
            end
        end

        table.sort(
            numbered,
            function(a, b)
                return a.idx < b.idx
            end
        )

        for _, entry in ipairs(numbered) do
            local letter =
                entry.obj.Letter.Text

            if type(letter) == "string" and
                letter ~= "" then

                table.insert(
                    letters,
                    letter
                )
            end
        end

        if #letters > 0 then
            return table.concat(letters):lower()
        end
    end

    local wordLabel =
        frame:FindFirstChild(
            "WordLabel",
            true
        )

    if not wordLabel then
        wordLabel =
            frame:FindFirstChild(
                "CurrentWordLabel",
                true
            )
    end

    if not wordLabel then
        wordLabel =
            frame:FindFirstChild(
                "WordDisplay",
                true
            )
    end

    if wordLabel and
        wordLabel:IsA("TextLabel") and
        wordLabel.Text ~= "" then

        return wordLabel.Text:lower()
    end

    local answerBox =
        FindAnswerBox(frame)

    if answerBox and
        answerBox.Text ~= "" then

        return answerBox.Text:lower()
    end

    return ""
end

--// REQUIRED LETTER
local function GetRequiredLetter()
    local frame = GetInGameFrame()

    if not frame then
        return nil
    end

    local requirementLabel =
        frame:FindFirstChild(
            "Requirement",
            true
        )

    if not requirementLabel then
        requirementLabel =
            frame:FindFirstChild(
                "RequiredLetter",
                true
            )
    end

    if not requirementLabel then
        requirementLabel =
            frame:FindFirstChild(
                "NextLetter",
                true
            )
    end

    if requirementLabel and
        requirementLabel:IsA("TextLabel") then

        local text = requirementLabel.Text

        local match =
            string.match(
                text,
                "([A-Za-z])"
            )

        if match then
            return match:lower()
        end
    end

    local wordLabel =
        frame:FindFirstChild(
            "LastWord",
            true
        )

    if not wordLabel then
        wordLabel =
            frame:FindFirstChild(
                "PreviousWord",
                true
            )
    end

    if wordLabel and
        wordLabel:IsA("TextLabel") then

        local word = wordLabel.Text

        if #word > 0 then
            local lastChar =
                string.sub(
                    word,
                    -1
                ):lower()

            if string.match(
                lastChar,
                "%a"
            ) then

                return lastChar
            end
        end
    end

    return nil
end

--// PREFIX RANGE
local function FindWordRange(prefix)
    local low = 1
    local high = #filteredDictionary

    local startIdx
    local endIdx

    while low <= high do
        local mid =
            math.floor(
                (low + high) / 2
            )

        if filteredDictionary[mid]:sub(
            1,
            #prefix
        ) < prefix then

            low = mid + 1
        else
            high = mid - 1
        end
    end

    startIdx = low

    low = startIdx
    high = #filteredDictionary

    while low <= high do
        local mid =
            math.floor(
                (low + high) / 2
            )

        if filteredDictionary[mid]:sub(
            1,
            #prefix
        ) == prefix then

            low = mid + 1
        else
            high = mid - 1
        end
    end

    endIdx = high

    if startIdx > endIdx then
        return nil, nil
    end

    return startIdx, endIdx
end

--// SUGGESTIONS
local function GetSuggestions(prefix)
    prefix = prefix:lower()

    if #prefix == 0 then
        return {}
    end

    local suggestions = {}

    local range =
        prefixCache[prefix]

    local sIdx
    local eIdx

    if range then
        sIdx = range[1]
        eIdx = range[2]
    else
        sIdx, eIdx =
            FindWordRange(prefix)

        if sIdx then
            prefixCache[prefix] = {
                sIdx,
                eIdx
            }

            local cacheCount = 0

            for _ in pairs(prefixCache) do
                cacheCount += 1

                if cacheCount >
                    MAX_PREFIX_CACHE then

                    prefixCache = {}
                    break
                end
            end
        end
    end

    if not sIdx then
        return {}
    end

    for i = sIdx,
        math.min(
            sIdx + SUGGESTION_COUNT * 3,
            eIdx
        ) do

        local word =
            filteredDictionary[i]

        if word and
            #suggestions <
                SUGGESTION_COUNT and
            word ~= prefix then

            if word:match(
                "^[a-z]+$"
            ) and #word >= 3 then

                if not usedWords[word] or
                    settings.showUsedWords then

                    table.insert(
                        suggestions,
                        word
                    )
                end
            end
        end
    end

    return suggestions
end

--// FORMAT
local function FormatSuggestionText(
    word,
    currentPrefix
)
    local isUsed =
        usedWords[word:lower()] or false

    local prefix =
        currentPrefix and
        currentPrefix:lower() or ""

    local lowerWord =
        word:lower()

    local result = ""

    if isUsed and
        not settings.showUsedWords then

        return ""
    end

    if prefix ~= "" and
        string.sub(
            lowerWord,
            1,
            #prefix
        ) == prefix then

        local typedPart =
            string.sub(
                word,
                1,
                #prefix
            )

        local remainingPart =
            string.sub(
                word,
                #prefix + 1
            )

        if isUsed and
            settings.showUsedWords then

            result =
                '<font color="#F2F2F2">' ..
                typedPart ..
                '</font>' ..

                '<font color="#888888">' ..
                remainingPart ..
                '</font>'
        else
            result =
                '<font color="#FFFFFF">' ..
                typedPart ..
                '</font>' ..
                remainingPart
        end
    else
        if isUsed and
            settings.showUsedWords then

            result =
                '<font color="#888888">' ..
                word ..
                '</font>'
        else
            result = word
        end
    end

    return result
end

local function LoadDictionaryFromTXT(url)
    local success, content =
        pcall(function()
            return game:HttpGet(url)
        end)

    if not success then
        return nil, 0
    end

    local words = {}
    local count = 0

    for line in content:gmatch(
        "[^\r\n]+"
    ) do
        local word =
            line:gsub(
                "%s+",
                ""
            ):lower()

        if word:match(
            "^[a-z]+$"
        ) and
            #word >= 3 and
            #word <= 20 then

            if not words[word] then
                words[word] = true
                count += 1
            end
        end
    end

    return words, count
end

--// DICTIONARY JSON
local function LoadDictionaryFromJSON(url)
    local success, content =
        pcall(function()
            return game:HttpGet(url)
        end)

    if not success then
        return nil, 0
    end

    local data

    local ok =
        pcall(function()
            data =
                HttpService:JSONDecode(
                    content
                )
        end)

    if not ok or
        type(data) ~= "table" then

        return nil, 0
    end

    local words = {}
    local count = 0

    for word in pairs(data) do
        if word:match(
            "^[a-zA-Z]+$"
        ) and
            #word >= 3 and
            #word <= 20 then

            local clean =
                word:lower()

            if not words[clean] then
                words[clean] = true
                count += 1
            end
        end
    end

    return words, count
end

--// LOAD ALL
local function LoadAllDictionaries()
    local wordMap = {}
    local totalWords = 0

    for _, source in ipairs(
        DICTIONARY_SOURCES
    ) do

        local words
        local count = 0

        if source.type == "txt" then
            words, count =
                LoadDictionaryFromTXT(
                    source.url
                )

        elseif source.type == "json" then
            words, count =
                LoadDictionaryFromJSON(
                    source.url
                )
        end

        if words and count > 0 then
            local newWords = 0

            for word in pairs(words) do
                if not wordMap[word] then
                    wordMap[word] = true
                    newWords += 1
                end
            end

            totalWords += newWords
            source.loaded = true
        end
    end

    if totalWords == 0 then
        local fallback = {
            "apple",

            "back",

            "bag",
            "ball",
            "band",
            "bank",

        }

    return totalWords
end

--// AUTO PLAY
local function AutoPlayLoop(token)
    while isAutoPlaying and
        isInitialized and
        token == autoPlayToken do

        task.wait(
            math.max(
                0.1,
                settings.autoPlayDelay
            )
        )

        if not isAutoPlaying then
            break
        end

        local requiredLetter =
            GetRequiredLetter()

        local currentWord =
            GetCurrentWord()

        local searchPrefix = ""

        if currentWord and
            #currentWord > 0 then

            searchPrefix =
                currentWord

        elseif requiredLetter and
            #requiredLetter > 0 then

            searchPrefix =
                requiredLetter

        else
            continue
        end

        local suggestions =
            GetSuggestions(
                searchPrefix
            )

        if #suggestions > 0 then
            local word =
                suggestions[
                    math.random(
                        1,
                        #suggestions
                    )
                ]

            if word then
                TypeWord(word)
            end
        end
    end
end

--// DRAGGING
local function MakeDraggable(target, handle)
    local dragging = false
    local dragStart
    local startPosition

    addConnection(
        handle.InputBegan:Connect(
            function(input)
                if input.UserInputType ==
                    Enum.UserInputType.MouseButton1
                    or
                    input.UserInputType ==
                    Enum.UserInputType.Touch then

                    dragging = true
                    dragStart =
                        input.Position

                    startPosition =
                        target.Position
                end
            end
        )
    )

    addConnection(
        handle.InputEnded:Connect(
            function(input)
                if input.UserInputType ==
                    Enum.UserInputType.MouseButton1
                    or
                    input.UserInputType ==
                    Enum.UserInputType.Touch then

                    dragging = false
                end
            end
        )
    )

    addConnection(
        UserInputService.InputChanged:Connect(
            function(input)
                if not dragging then
                    return
                end

                if input.UserInputType ~=
                    Enum.UserInputType.MouseMovement
                    and
                    input.UserInputType ~=
                        Enum.UserInputType.Touch then

                    return
                end

                local delta =
                    input.Position -
                    dragStart

                target.Position =
                    UDim2.new(
                        startPosition.X.Scale,
                        startPosition.X.Offset +
                            delta.X,

                        startPosition.Y.Scale,
                        startPosition.Y.Offset +
                            delta.Y
                    )
            end
        )
    )
end

--// UPDATE STATS
local function UpdateStats()
    if not statsTyped then
        return
    end

    statsTyped.Text =
        formatNumber(wordsTyped)

    statsUsed.Text =
        formatNumber(
            0 +
            (function()
                local count = 0

                for _ in pairs(usedWords) do
                    count += 1
                end

                return count
            end)()
        )

    statsPool.Text =
        formatNumber(
            #filteredDictionary
        )

    local elapsed =
        math.max(
            1,
            os.clock() - sessionStart
        )

    local calculatedWpm =
        wordsTyped /
        elapsed *
        60

    statsWpm.Text =
        string.format(
            "%.0f",
            calculatedWpm
        )
end

--// UPDATE SUGGESTIONS
UpdateSuggestions = function()
    if not isInitialized or
        #filteredDictionary == 0 then

        if statusLabel then
            statusLabel.Text =
                "dictionary loading..."
        end

        return
    end

    local requiredLetter =
        GetRequiredLetter()

    local searchPrefix = ""

    local currentWord =
        GetCurrentWord()

    if currentWord and
        #currentWord > 0 then

        searchPrefix =
            currentWord

    elseif requiredLetter and
        #requiredLetter > 0 then

        searchPrefix =
            requiredLetter

    else
        for _, button in ipairs(
            suggestionButtons
        ) do

            button.Visible = false
        end

        if statusLabel then
            statusLabel.Text = "ready"
        end

        if prefixValue then
            prefixValue.Text = "—"
        end

        if wordCountLabel then
            wordCountLabel.Text =
                formatNumber(
                    #filteredDictionary
                ) ..
                " words"
        end

        return
    end

    currentPrefix = searchPrefix

    prefixValue.Text =
        '"' ..
        searchPrefix ..
        '"'

    local suggestions =
        GetSuggestions(
            searchPrefix
        )

    for i, button in ipairs(
        suggestionButtons
    ) do

        if suggestions[i] then
            local formatted =
                FormatSuggestionText(
                    suggestions[i],
                    searchPrefix
                )

            if formatted ~= "" then
                button.Text = formatted
                button.Visible = true

                button:SetAttribute(
                    "Word",
                    suggestions[i]
                )
            else
                button.Visible = false
                button:SetAttribute(
                    "Word",
                    ""
                )
            end
        else
            button.Visible = false

            button:SetAttribute(
                "Word",
                ""
            )
        end
    end

    local visibleCount = 0

    for _, button in ipairs(
        suggestionButtons
    ) do

        if button.Visible then
            visibleCount += 1
        end
    end

    suggestionsFrame.CanvasSize =
        UDim2.new(
            0,
            0,
            0,
            visibleCount * 37 + 10
        )

    if #suggestions > 0 then
        statusLabel.Text =
            "found " ..
            #suggestions ..
            " matches"
    else
        statusLabel.Text =
            "no new words"
    end

    wordCountLabel.Text =
        formatNumber(
            #filteredDictionary
        ) ..
        " words"

    lastSuggestedPrefix =
        searchPrefix

    UpdateStats()
end

--// GAME GUI TRIGGERS
local function SetupTriggers()
    local inGameGui =
        playerGui:FindFirstChild(
            "InGame"
        )

    if not inGameGui then
        return
    end

    timeCounterRunning = true

    local frame =
        inGameGui:FindFirstChild(
            "Frame"
        )

    if not frame then
        return
    end

    local wordFrame =
        frame:FindFirstChild(
            "CurrentWord"
        )

    if wordFrame then
        addConnection(
            wordFrame.ChildAdded:Connect(
                function()
                    task.wait(0.04)
                    UpdateSuggestions()
                end
            )
        )

        addConnection(
            wordFrame.ChildRemoved:Connect(
                function()
                    task.wait(0.04)
                    UpdateSuggestions()
                end
            )
        )
    end

    local box =
        FindAnswerBox(frame)

    if box then
        addConnection(
            box:GetPropertyChangedSignal(
                "Text"
            ):Connect(
                function()
                    UpdateSuggestions()
                end
            )
        )
    end

    addConnection(
        inGameGui.DescendantAdded:Connect(
            function(object)
                if object:IsA("TextBox") or
                    object:IsA("TextLabel") then

                    task.defer(
                        UpdateSuggestions
                    )
                end
            end
        )
    )
end

--// DETECT IN-GAME GUI
addConnection(
    playerGui.ChildAdded:Connect(
        function(child)
            if child.Name ==
                "InGame" then

                task.wait(0.15)

                SetupTriggers()
                UpdateSuggestions()
            end
        end
    )
)

            task.wait(1)

            if not guiDestroyed then
                SetupTriggers()
                UpdateSuggestions()
            end
        end
    )
)
